  kubectl -n {{ .Release.Namespace }} exec -it deploy/{{ include "haben.fullname" . }}-restore -- restore.sh
  kubectl -n {{ .Release.Namespace }} exec -it deploy/{{ include "haben.fullname" . }}-restore -- restore.sh latest
